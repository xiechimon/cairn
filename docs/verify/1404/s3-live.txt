== dump.mjs
// Usage: node dump.mjs <server.db> — prints migrations, gate tables' schema and indexes.
import { createRequire } from 'node:module';
const require = createRequire('/home/user/pacman/.claude/worktrees/p3c/apps/server/package.json');
const Database = require('better-sqlite3');
const db = new Database(process.argv[2], { readonly: true });
const mig = db.prepare('SELECT id, hash, created_at FROM __drizzle_migrations ORDER BY created_at').all();
console.log(`__drizzle_migrations rows: ${mig.length}; newest created_at: ${mig.at(-1)?.created_at}`);
const tables = db.prepare("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' AND name <> '__drizzle_migrations' ORDER BY name").all().map((r) => r.name);
console.log(`tables (${tables.length}): ${tables.join(' ')}`);
for (const t of ['gate_digest', 'gate_round', 'gate_verdict', 'gate_state']) {
  const row = db.prepare("SELECT sql FROM sqlite_master WHERE type='table' AND name=?").get(t);
  console.log(`\n-- ${t}: ${row ? '' : 'ABSENT'}`);
  if (!row) continue;
  console.log(`${row.sql};`);
  for (const ix of db.prepare("SELECT name, sql FROM sqlite_master WHERE type='index' AND tbl_name=? ORDER BY name").all(t)) {
    console.log(ix.sql ? `${ix.sql};` : `-- ${ix.name} (implicit, from PRIMARY KEY/UNIQUE)`);
  }
  console.log(`-- index_list: ${JSON.stringify(db.prepare(`PRAGMA index_list(${t})`).all())}`);
  console.log(`-- foreign_key_list: ${JSON.stringify(db.prepare(`PRAGMA foreign_key_list(${t})`).all().map((f) => ({ from: f.from, table: f.table, to: f.to, on_delete: f.on_delete })))}`);
}
db.close();

== probe.mjs
// Usage: node probe.mjs <seed|dups|counts> <server.db> <buildId>
// Opens its own connection with foreign_keys = ON (a per-connection pragma, as db/client.ts sets it).
import { createRequire } from 'node:module';
const require = createRequire('/home/user/pacman/.claude/worktrees/p3c/apps/server/package.json');
const Database = require('better-sqlite3');
const [cmd, dbPath, buildId] = process.argv.slice(2);
const db = new Database(dbPath);
db.pragma('foreign_keys = ON');
db.pragma('busy_timeout = 5000');
const D1 = `pd2:${'a'.repeat(64)}`;
const D2 = `pd2:${'b'.repeat(64)}`;
const H1 = '1'.repeat(40);
const run = (label, sql, ...args) => {
  try {
    const r = db.prepare(sql).run(...args);
    console.log(`${label}: ok (changes=${r.changes})`);
  } catch (e) {
    console.log(`${label}: ${e.code} ${e.message}`);
  }
};
const counts = () => {
  const out = {};
  for (const t of ['gate_digest', 'gate_round', 'gate_verdict', 'gate_state']) {
    out[t] = db.prepare(`SELECT count(*) AS n FROM ${t} WHERE buildId = ?`).get(buildId).n;
  }
  out.build = db.prepare('SELECT count(*) AS n FROM build WHERE id = ?').get(buildId).n;
  out.step = db.prepare('SELECT count(*) AS n FROM step WHERE buildId = ?').get(buildId).n;
  return out;
};
const stepId = () => db.prepare('SELECT id FROM step WHERE buildId = ? ORDER BY createdAt LIMIT 1').get(buildId)?.id;
if (cmd === 'seed') {
  const s = stepId();
  console.log(`build ${buildId}, step ${s}`);
  run('INSERT gate_digest (buildId, headSha=H1, pd2:a…)', 'INSERT INTO gate_digest (buildId, headSha, patchDigest, countedFiles, countedLines, keptFiles, keptLines, createdAt) VALUES (?, ?, ?, 3, 42, 2, 40, ?)', buildId, H1, D1, Date.now());
  run('INSERT gate_round (round 1, pd2:a…)', 'INSERT INTO gate_round (id, buildId, patchDigest, round, perspectives, startedAt) VALUES (?, ?, ?, 1, ?, ?)', `r1-${buildId}`, buildId, D1, JSON.stringify(['code', 'security', 'tests']), Date.now());
  run('INSERT gate_verdict (on the build\'s step)', "INSERT INTO gate_verdict (stepId, buildId, kind, perspective, outcome, patchDigest, headSha, expectedHeadSha, createdAt) VALUES (?, ?, 'review', 'security', 'pass', ?, ?, ?, ?)", s, buildId, D1, H1, H1, Date.now());
  run('INSERT gate_state', "INSERT INTO gate_state (buildId, ready, headSha, patchDigest, policy, blockers, evaluatedAt) VALUES (?, 0, ?, ?, 'human', ?, ?)", buildId, H1, D1, JSON.stringify([{ code: 'policy-human', class: 'policy', label: '项目策略为人工合并', detail: [] }]), Date.now());
  console.log('row defaults:', JSON.stringify(db.prepare('SELECT round, fixRound, state, perspectives, timedOut, dispatches FROM gate_round WHERE buildId = ?').get(buildId)));
  console.log('counts:', JSON.stringify(counts()));
} else if (cmd === 'dups') {
  const s = stepId();
  run('duplicate gate_round (same buildId, same patchDigest, round 2)', 'INSERT INTO gate_round (id, buildId, patchDigest, round, perspectives, startedAt) VALUES (?, ?, ?, 2, ?, ?)', `r2-${buildId}`, buildId, D1, '[]', Date.now());
  run('duplicate gate_round number (same buildId, other digest, round 1)', 'INSERT INTO gate_round (id, buildId, patchDigest, round, perspectives, startedAt) VALUES (?, ?, ?, 1, ?, ?)', `r3-${buildId}`, buildId, D2, '[]', Date.now());
  run('second gate_round (same buildId, other digest, round 2)', 'INSERT INTO gate_round (id, buildId, patchDigest, round, perspectives, startedAt) VALUES (?, ?, ?, 2, ?, ?)', `r4-${buildId}`, buildId, D2, '[]', Date.now());
  run('duplicate gate_verdict (same stepId)', "INSERT INTO gate_verdict (stepId, buildId, kind, outcome, expectedHeadSha, createdAt) VALUES (?, ?, 'review', 'block', ?, ?)", s, buildId, H1, Date.now());
  run('gate_verdict for a step that does not exist', "INSERT INTO gate_verdict (stepId, buildId, kind, outcome, expectedHeadSha, createdAt) VALUES ('no-such-step', ?, 'review', 'pass', ?, ?)", buildId, H1, Date.now());
  run('duplicate gate_state (same buildId)', "INSERT INTO gate_state (buildId, ready, policy, blockers, evaluatedAt) VALUES (?, 1, 'human', '[]', ?)", buildId, Date.now());
  run('duplicate gate_digest (same buildId, same headSha)', 'INSERT INTO gate_digest (buildId, headSha, patchDigest, countedFiles, countedLines, keptFiles, keptLines, createdAt) VALUES (?, ?, ?, 1, 1, 1, 1, ?)', buildId, H1, D2, Date.now());
  console.log('counts:', JSON.stringify(counts()));
} else if (cmd === 'counts') {
  console.log('counts:', JSON.stringify(counts()));
}
db.close();
