    const count = src.split(oldText).length - 1;
    if (count !== 1) {
      rows.push({ name, summary: `NOT APPLIED (${count} matches)`, failed: [], patch: '' });
      continue;
    }
    writeFileSync(path, src.replace(oldText, () => newText));
    try {
      const patch = git('diff', '--no-color', '--', file);
      const r = spawnSync('pnpm', ['exec', 'vitest', 'run', ...tests], { cwd: root, encoding: 'utf8' });
      const out = `${r.stdout}\n${r.stderr}`.replace(/\x1b\[[0-9;]*m/g, '');
      const line = out.split('\n').find((l) => /^\s+Tests\s/.test(l)) ?? 'no summary';
      const failed = out
        .split('\n')
        .filter((l) => /^\s+× /.test(l))
        .map((l) => l.trim().replace(/ \d+ms$/, ''));
      rows.push({ name, summary: line.replace(/^\s+Tests\s+/, '').trim(), failed, patch });
    } finally {
      writeFileSync(path, src);
    }
  }
  const after = git('diff');
  for (const { name, summary } of rows) console.log(`${name.padEnd(52)} ${summary}`);
  console.log('');
  console.log(`git diff identical before and after the run: ${before === after}`);
  console.log(`survivors: ${rows.filter((r) => r.failed.length === 0).length}`);
  console.log('');
  console.log('per mutation: patch, then the failing tests');

    console.log('');

    console.log(patch.trimEnd());
    console.log('failing tests:');
    for (const f of failed) console.log(`    ${f}`);
