  import hashlib
  import json
  import sys
  
  # Written from ADR 0021 D3 as amended on PR #1410 (origin/claude/nice-faraday-o329b3):
  # exclude docs/verify/** (a rename only when both names are under it), take the JSON tuple
  # [status, filename, previous_filename, blobSha, patchSha] per file, sort, sha256.
  # patchSha = sha256 (hex) of the file's compare `patch` text, null when there is no patch.
  # Not specified by D3 and taken from the earlier crosscheck (digest-crosscheck.txt): compact
  # JSON, tuples joined by "\n", UTF-8; the version prefix is "pd2" (PATCH_DIGEST_VERSION).
  PREFIX = "pd2"
  
  
  def sha256_hex(text):
      return hashlib.sha256(text.encode("utf-8")).hexdigest()
  
  
  def patch_sha(patch):
      return sha256_hex(patch) if isinstance(patch, str) and patch != "" else None
  
  
  def is_evidence(path):
      return path.startswith("docs/verify/")
  
  
  def digest(entries):
      kept = [e for e in entries
              if not (is_evidence(e["filename"])
                      and (e.get("previous_filename") is None or is_evidence(e["previous_filename"])))]
      tuples = sorted(json.dumps([e["status"], e["filename"], e.get("previous_filename"), e.get("sha"),
                                  patch_sha(e.get("patch"))],
                                 separators=(",", ":"), ensure_ascii=False) for e in kept)
      return kept, PREFIX + ":" + hashlib.sha256("\n".join(tuples).encode("utf-8")).hexdigest()
  
  
  fixture = json.load(open(sys.argv[1], encoding="utf-8"))
  files = fixture["files"]
  kept, d = digest(files)
  print("fixture:", len(files), "files,", len(kept), "kept,",
        sum(1 for e in kept if patch_sha(e.get("patch")) is None), "kept without patch")
  print("fixture digest:", d)
  print("patchSha('(patch trimmed)'):", sha256_hex("(patch trimmed)"))
  print("patchSha('@@ -1 +1 @@\\n-a\\n+b'):", sha256_hex("@@ -1 +1 @@\n-a\n+b"))
  print("patchSha('@@ -1 +1 @@\\n-旧\\n+新'):", sha256_hex("@@ -1 +1 @@\n-旧\n+新"))
  
  
  # Synthetic vectors used by the unit tests. They give patchSha directly (a test stand-in of
  # 64 'c'), so these tuples are built without the patch_sha step.
  def digest_tuples(rows):
      tuples = sorted(json.dumps(list(r), separators=(",", ":"), ensure_ascii=False) for r in rows)
      for t in tuples:
          print("  " + t)
      return PREFIX + ":" + hashlib.sha256("\n".join(tuples).encode("utf-8")).hexdigest()
  
  
  C = "c" * 64
  print("code-unit sort test:")
  print(digest_tuples([
      ("modified", "apps/web/src/a/b.ts", None, "1" * 40, C),
      ("modified", "apps/web/src/a.b.ts", None, "2" * 40, C),
      ("modified", "apps/web/src/B.ts", None, "3" * 40, C),
      ("modified", "apps/web/src/a-b.ts", None, "4" * 40, C),
  ]))
  print("null patchSha test:")
  print(digest_tuples([
      ("modified", "apps/server/src/a.ts", None, "1" * 40, C),
      ("added", "apps/web/public/logo.png", None, "5" * 40, None),
  ]))
  
  # The review's scenario, as compare entries: base f=A, PR f=B, main f=C. Before the PR merges
  # main the three-dot compare is A -> B; after merging main and keeping B it is C -> B. Same
  # status, filename and head blob; only the patch text differs.
  entry = {"status": "modified", "filename": "apps/server/src/f.ts", "sha": "b" * 40,
           "additions": 1, "deletions": 1}
  _, before = digest([dict(entry, patch="@@ -1 +1 @@\n-A\n+B")])
  _, after = digest([dict(entry, patch="@@ -1 +1 @@\n-C\n+B")])
  print("scenario before merge (A -> B):", before)
  print("scenario after merge  (C -> B):", after)
  print("scenario digests differ:", before != after)
