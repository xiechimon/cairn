def sh(cmd, check=True, cwd=WT):
    r = subprocess.run(cmd, cwd=cwd, shell=True, capture_output=True, text=True)
    if check and r.returncode != 0:
        sys.exit(f'command failed: {cmd}\n{r.stdout}\n{r.stderr}')
    return r

def path(p):
    return os.path.join(WT, p)

def restore():
    for f in TRACKED:
        open(path(f), 'w', encoding='utf8').write(sh(f'git show HEAD:{f}').stdout)
    if os.path.exists(path(LATER)):
        os.remove(path(LATER))
    sh('git diff --exit-code HEAD -- apps/server')
    if sh('git status --porcelain -- apps/server').stdout.strip():
        sys.exit('worktree not clean after restore')

def regenerate():

    return (summary.group(1).strip() if summary else 'no summary'), failed

// Usage: node s3r2-recreate-sql.mjs <table> [keep]
// Prints a drizzle-kit-shaped table recreate for <table> (the shape of 0004_opposite_chimera.sql), built
// from the CREATE statement and columns of a database migrated with the worktree's drizzle folder.
// With `keep`, the four gate tables are copied aside before the recreate and written back after it.
import { createRequire } from 'node:module';
const WT = '/home/user/pacman/.claude/worktrees/p3c';
const require = createRequire(`${WT}/apps/server/package.json`);
const Database = require('better-sqlite3');
const { drizzle } = require('drizzle-orm/better-sqlite3');
const { migrate } = require('drizzle-orm/better-sqlite3/migrator');
const [table, keep] = process.argv.slice(2);
const s = new Database(':memory:');
migrate(drizzle(s), { migrationsFolder: `${WT}/apps/server/drizzle` });
const { sql } = s.prepare("SELECT sql FROM sqlite_master WHERE type = 'table' AND name = ?").get(table);
const cols = s.prepare(`PRAGMA table_info(${table})`).all().map((c) => `"${c.name}"`).join(', ');
const indexes = s.prepare("SELECT sql FROM sqlite_master WHERE type = 'index' AND tbl_name = ? AND sql IS NOT NULL").all(table).map((r) => `${r.sql};`);
const GATE = ['gate_digest', 'gate_round', 'gate_verdict', 'gate_state'];
const stmts = [
  ...(keep ? GATE.map((t) => `CREATE TABLE \`__keep_${t}\` AS SELECT * FROM \`${t}\`;`) : []),
  'PRAGMA foreign_keys=OFF;',
  `${sql.replace(new RegExp(`^CREATE TABLE [\`"]?${table}[\`"]?`), `CREATE TABLE \`__new_${table}\``)};`,
  `INSERT INTO \`__new_${table}\`(${cols}) SELECT ${cols} FROM \`${table}\`;`,
  `DROP TABLE \`${table}\`;`,
  `ALTER TABLE \`__new_${table}\` RENAME TO \`${table}\`;`,
  'PRAGMA foreign_keys=ON;',
  ...indexes,
  ...(keep ? GATE.flatMap((t) => [`INSERT INTO \`${t}\` SELECT * FROM \`__keep_${t}\`;`, `DROP TABLE \`__keep_${t}\`;`]) : []),
];
process.stdout.write(stmts.join('--> statement-breakpoint\n'));
