== s3-mutate.py
# Applies each mutation alone, regenerates the migration from the mutated schema.ts the way the
# repo does (drizzle-kit generate --name merge_gate_tables, after dropping 0044 and restoring the
# journal to origin/main's), runs the S3 test file, then restores every touched file from HEAD.
import json, os, re, subprocess, sys

WT = '/home/user/pacman/.claude/worktrees/p3c'
SCHEMA = 'apps/server/src/db/schema.ts'
JOURNAL = 'apps/server/drizzle/meta/_journal.json'
SQL = 'apps/server/drizzle/0044_merge_gate_tables.sql'
SNAP = 'apps/server/drizzle/meta/0044_snapshot.json'
TEST = 'apps/server/test/merge-gate-tables.test.ts'
TOUCHED = [SCHEMA, JOURNAL, SQL, SNAP]

MUTATIONS = [
    ('gate_round: UNIQUE (buildId, patchDigest) dropped', SCHEMA,
     "    uniqueIndex('gate_round_build_digest_unique').on(t.buildId, t.patchDigest),\n", ''),
    ('gate_round: UNIQUE (buildId, round) dropped', SCHEMA,
     "    uniqueIndex('gate_round_build_round_unique').on(t.buildId, t.round),\n", ''),
    ('gate_verdict: patchDigest NOT NULL', SCHEMA,
     "    patchDigest: text('patchDigest'),\n    /** 判定自报的 head",
     "    patchDigest: text('patchDigest').notNull(),\n    /** 判定自报的 head"),
    ('delete: gate_verdict.buildId FK without cascade', SCHEMA,
     "    buildId: text('buildId')\n      .notNull()\n      .references(() => build.id, { onDelete: 'cascade' }),\n    kind:",
     "    buildId: text('buildId')\n      .notNull()\n      .references(() => build.id),\n    kind:"),
    ('delete: gate_state.buildId with no FK', SCHEMA,
     "  buildId: text('buildId')\n    .primaryKey()\n    .references(() => build.id, { onDelete: 'cascade' }),\n  /** MergeReadiness.ready",
     "  buildId: text('buildId').primaryKey(),\n  /** MergeReadiness.ready"),
    ('delete: gate_verdict.stepId FK without cascade', SCHEMA,
     "      .references(() => step.id, { onDelete: 'cascade' }),",
     "      .references(() => step.id),"),
    ('gate_verdict: stepId FK dropped', SCHEMA,
     "    stepId: text('stepId')\n      .primaryKey()\n      .references(() => step.id, { onDelete: 'cascade' }),",
     "    stepId: text('stepId').primaryKey(),"),
    ('journal: 0044 when below 0043', JOURNAL, None, None),
]

def sh(cmd, check=True):
    r = subprocess.run(cmd, cwd=WT, shell=True, capture_output=True, text=True)
    if check and r.returncode != 0:
        sys.exit(f'command failed: {cmd}\n{r.stdout}\n{r.stderr}')
    return r

def restore():
    for f in TOUCHED:
        data = sh(f'git show HEAD:{f}').stdout
        open(os.path.join(WT, f), 'w', encoding='utf8').write(data)
    sh('git diff --exit-code HEAD -- apps/server')
    if sh('git status --porcelain -- apps/server').stdout.strip():
        sys.exit('worktree not clean after restore')

def regenerate():
    os.remove(os.path.join(WT, SQL))
    os.remove(os.path.join(WT, SNAP))
    open(os.path.join(WT, JOURNAL), 'w', encoding='utf8').write(sh(f'git show origin/main:{JOURNAL}').stdout)
    sh('cd apps/server && ./node_modules/.bin/drizzle-kit generate --name merge_gate_tables')

def run_tests():
    r = sh(f'pnpm exec vitest run {TEST} --reporter=verbose', check=False)
    out = r.stdout + r.stderr
    summary = re.search(r'Tests\s+(.*?)\n', out)
    failed = sorted(set(m.group(1).strip() for m in re.finditer(r'^\s*×\s+(.*?)(?:\s+\d+ms)?$', out, re.M)))
    return (summary.group(1).strip() if summary else 'no summary'), failed

rows = []
for name, path, old, new in MUTATIONS:
    if path == JOURNAL:
        j = json.load(open(os.path.join(WT, JOURNAL), encoding='utf8'))
        prev = next(e for e in j['entries'] if e['tag'].startswith('0043'))
        mine = next(e for e in j['entries'] if e['tag'].startswith('0044'))
        mine['when'] = prev['when'] - 1
        open(os.path.join(WT, JOURNAL), 'w', encoding='utf8').write(json.dumps(j, indent=2) + '\n')
        diff_desc = f'{JOURNAL}: 0044 when -> {prev["when"] - 1} (0043 when - 1)'
    else:
        src = open(os.path.join(WT, path), encoding='utf8').read()
        if src.count(old) != 1:
            sys.exit(f'{name}: anchor found {src.count(old)} times')
        open(os.path.join(WT, path), 'w', encoding='utf8').write(src.replace(old, new))
        regenerate()
        diff_desc = sh(f'git diff --no-color -U0 HEAD -- {SQL}', check=False).stdout
        diff_desc = '\n'.join(l for l in diff_desc.splitlines() if l[:1] in '+-' and not l.startswith(('+++', '---')))
    summary, failed = run_tests()
    rows.append((name, summary, failed, diff_desc))
    restore()

for name, summary, failed, diff_desc in rows:
    print(f'{name:<52} {summary}')
print()
for name, summary, failed, diff_desc in rows:
    print(f'## {name}')
    print('migration change:')
    print(diff_desc)
    print('red tests:')
    for t in failed:
        print(f'  {t}')
    print()
