crosscheck3.py:
  import hashlib
  import json
  import sys
  
  # Round 6 reference for the #1404 S1 patch digest (ADR 0021 D3 as amended on PR #1410): exclude
  # docs/verify/** (a rename only when both names are under it), take the JSON tuple
  # [status, filename, previous_filename, blobSha, patchSha] per file, sort, sha256 over the tuples
  # joined by "\n" in UTF-8, prefix "pd2:". patchSha = sha256 (hex) of the compare `patch` text, null
  # when there is no patch.
  #
  # Change from crosscheck2.py: the sort key. digest.ts uses JS's default Array.prototype.sort, which
  # compares UTF-16 code units; Python's sorted() on str compares code points. The two differ only when
  # a supplementary-plane character (a surrogate pair, 0xD800-0xDBFF lead unit) meets a BMP character in
  # U+E000-U+FFFF. Encoding to UTF-16BE turns each code unit into two big-endian bytes, so comparing
  # the bytes compares the code units.
  PREFIX = "pd2"
  
  
  def utf16_key(s):
      return s.encode("utf-16-be")
  
  
  def sha256_hex(text):
      return hashlib.sha256(text.encode("utf-8")).hexdigest()
  
  
  def patch_sha(patch):
      return sha256_hex(patch) if isinstance(patch, str) and patch != "" else None
  
  
  def is_evidence(path):
      return path.startswith("docs/verify/")
  
  
  def tuple_json(row):
      return json.dumps(list(row), separators=(",", ":"), ensure_ascii=False)
  
  
  def digest_of(tuples, key):
      ordered = sorted(tuples, key=key)
      return ordered, PREFIX + ":" + hashlib.sha256("\n".join(ordered).encode("utf-8")).hexdigest()
  
  
  def fixture_digest(entries, key):
      kept = [e for e in entries
              if not (is_evidence(e["filename"])
                      and (e.get("previous_filename") is None or is_evidence(e["previous_filename"])))]
      tuples = [tuple_json([e["status"], e["filename"], e.get("previous_filename"), e.get("sha"),
                            patch_sha(e.get("patch"))]) for e in kept]
      return kept, digest_of(tuples, key)[1]
  
  
  def show(label, rows):
      tuples = [tuple_json(r) for r in rows]
      by_unit, d_unit = digest_of(tuples, utf16_key)
      by_point, d_point = digest_of(tuples, None)
      print(label)
      print("  UTF-16 code-unit order:")
      for t in by_unit:
          print("    " + t.encode("ascii", "backslashreplace").decode("ascii"))
      print("  " + d_unit)
      print("  code-point order:")
      for t in by_point:
          print("    " + t.encode("ascii", "backslashreplace").decode("ascii"))
      print("  " + d_point)
      print("  orders differ:", by_unit != by_point)
      return d_unit
  
  
  fixture = json.load(open(sys.argv[1], encoding="utf-8"))
  files = fixture["files"]
  kept, d_unit = fixture_digest(files, utf16_key)
  _, d_point = fixture_digest(files, None)
  print("fixture:", len(files), "files,", len(kept), "kept")
  print("fixture digest, UTF-16 code-unit sort:", d_unit)
  print("fixture digest, code-point sort:      ", d_point)
  print()
  
  C = "c" * 64
  # U+1F600 is the surrogate pair D83D DE00; U+FF21 (fullwidth A) is one unit, 0xFF21. By code unit
  # 0xD83D < 0xFF21, so the emoji name sorts first; by code point 0x1F600 > 0xFF21, so it sorts last.
  show("non-BMP test (emoji U+1F600 next to U+FF21):", [
      ("modified", "apps/web/src/\uFF21.ts", None, "1" * 40, C),
      ("modified", "apps/web/src/\U0001F600.ts", None, "2" * 40, C),
  ])
  show("code-unit sort test (round 5 vector, ASCII):", [
      ("modified", "apps/web/src/a/b.ts", None, "1" * 40, C),
      ("modified", "apps/web/src/a.b.ts", None, "2" * 40, C),
      ("modified", "apps/web/src/B.ts", None, "3" * 40, C),
      ("modified", "apps/web/src/a-b.ts", None, "4" * 40, C),
  ])
  show("null patchSha test (round 5 vector):", [
      ("modified", "apps/server/src/a.ts", None, "1" * 40, C),
      ("added", "apps/web/public/logo.png", None, "5" * 40, None),
  ])
