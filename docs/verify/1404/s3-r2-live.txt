== s3r2-dump.mjs
// Usage: node s3r2-dump.mjs <server.db> — migrations, table list, gate_verdict DDL, foreign keys of the gate tables.
import { createRequire } from 'node:module';
const require = createRequire('/home/user/pacman/.claude/worktrees/p3c/apps/server/package.json');
const Database = require('better-sqlite3');
const db = new Database(process.argv[2], { readonly: true });
const mig = db.prepare('SELECT created_at FROM __drizzle_migrations ORDER BY created_at').all();
console.log(`__drizzle_migrations rows: ${mig.length}; newest created_at: ${mig.at(-1)?.created_at}`);
const tables = db.prepare("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' AND name <> '__drizzle_migrations' ORDER BY name").all().map((r) => r.name);
console.log(`tables (${tables.length}); gate tables: ${tables.filter((t) => t.startsWith('gate_')).join(' ') || 'none'}`);
const v = db.prepare("SELECT sql FROM sqlite_master WHERE type='table' AND name='gate_verdict'").get();
if (v) console.log(`-- gate_verdict DDL:\n${v.sql};`);
for (const t of ['gate_digest', 'gate_round', 'gate_verdict', 'gate_state']) {
  if (!tables.includes(t)) continue;
  console.log(`-- ${t} foreign_key_list: ${JSON.stringify(db.prepare(`PRAGMA foreign_key_list(${t})`).all().map((f) => ({ from: f.from, table: f.table, to: f.to, on_delete: f.on_delete })))}`);
}
const refs = tables.filter((t) => db.prepare(`PRAGMA foreign_key_list(${t})`).all().some((f) => f.table === 'step'));
console.log(`tables with a foreign key to step: ${refs.length ? refs.join(' ') : 'none'}`);
db.close();

== s3r2-probe.mjs
// Usage: node s3r2-probe.mjs <seed|stepdel|counts> <server.db> <buildId>
// Own connection with foreign_keys = ON (per-connection pragma, as db/client.ts sets it).
import { createRequire } from 'node:module';
const require = createRequire('/home/user/pacman/.claude/worktrees/p3c/apps/server/package.json');
const Database = require('better-sqlite3');
const [cmd, dbPath, buildId] = process.argv.slice(2);
const db = new Database(dbPath);
db.pragma('foreign_keys = ON');
db.pragma('busy_timeout = 5000');
const D1 = `pd2:${'a'.repeat(64)}`;
const H1 = '1'.repeat(40);
const run = (label, sql, ...args) => {
  try {
    console.log(`${label}: ok (changes=${db.prepare(sql).run(...args).changes})`);
  } catch (e) {
    console.log(`${label}: ${e.code} ${e.message}`);
  }
};
const counts = () => {
  const out = {};
  for (const t of ['gate_digest', 'gate_round', 'gate_verdict', 'gate_state']) {
    out[t] = db.prepare(`SELECT count(*) AS n FROM ${t} WHERE buildId = ?`).get(buildId).n;
  }
  out.build = db.prepare('SELECT count(*) AS n FROM build WHERE id = ?').get(buildId).n;
  out.step = db.prepare('SELECT count(*) AS n FROM step WHERE buildId = ?').get(buildId).n;
  return out;
};
const firstStep = () => db.prepare('SELECT id FROM step WHERE buildId = ? ORDER BY createdAt LIMIT 1').get(buildId)?.id;
if (cmd === 'seed') {
  const s = firstStep();
  console.log(`build ${buildId}, step ${s}`);
  run('INSERT gate_digest', 'INSERT INTO gate_digest (buildId, headSha, patchDigest, countedFiles, countedLines, keptFiles, keptLines, createdAt) VALUES (?, ?, ?, 3, 42, 2, 40, ?)', buildId, H1, D1, Date.now());
  run('INSERT gate_round', 'INSERT INTO gate_round (id, buildId, patchDigest, round, perspectives, startedAt) VALUES (?, ?, ?, 1, ?, ?)', `r1-${buildId}`, buildId, D1, '["code","security","tests"]', Date.now());
  run("INSERT gate_verdict on the build's step (outcome block)", "INSERT INTO gate_verdict (stepId, buildId, kind, perspective, outcome, patchDigest, headSha, expectedHeadSha, createdAt) VALUES (?, ?, 'review', 'security', 'block', ?, ?, ?, ?)", s, buildId, D1, H1, H1, Date.now());
  run('INSERT gate_verdict whose stepId has no step row', "INSERT INTO gate_verdict (stepId, buildId, kind, outcome, expectedHeadSha, createdAt) VALUES ('no-such-step', ?, 'verify', 'pass', ?, ?)", buildId, H1, Date.now());
  run('INSERT gate_verdict for a build that does not exist', "INSERT INTO gate_verdict (stepId, buildId, kind, outcome, expectedHeadSha, createdAt) VALUES ('s-x', 'no-such-build', 'review', 'pass', ?, ?)", H1, Date.now());
  run('INSERT gate_verdict with expectedHeadSha null', "INSERT INTO gate_verdict (stepId, buildId, kind, outcome, expectedHeadSha, createdAt) VALUES ('s-y', ?, 'review', 'pass', NULL, ?)", buildId, Date.now());
  run('INSERT gate_round with patchDigest null', "INSERT INTO gate_round (id, buildId, patchDigest, round, perspectives, startedAt) VALUES ('r-null', ?, NULL, 2, '[]', ?)", buildId, Date.now());
  run('INSERT gate_state', "INSERT INTO gate_state (buildId, ready, headSha, patchDigest, policy, blockers, evaluatedAt) VALUES (?, 0, ?, ?, 'human', '[]', ?)", buildId, H1, D1, Date.now());
  console.log('counts:', JSON.stringify(counts()));
} else if (cmd === 'stepdel') {
  const s = firstStep();
  run(`DELETE FROM step WHERE id = ${s} (the build stays)`, 'DELETE FROM step WHERE id = ?', s);
  console.log('verdicts left:', JSON.stringify(db.prepare('SELECT stepId, outcome FROM gate_verdict WHERE buildId = ? ORDER BY stepId').all(buildId)));
  console.log('counts:', JSON.stringify(counts()));
} else if (cmd === 'counts') {
  console.log('counts:', JSON.stringify(counts()));
}
db.close();
