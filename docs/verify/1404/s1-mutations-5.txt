  // Mutation runner for #1404 S1 round 5. Each mutation replaces text in one source file, runs the three S1 test
  // files, restores the file. Usage: node mutate.mjs <worktree> <mutations.json>
  import { execFileSync, spawnSync } from 'node:child_process';
  import { readFileSync, writeFileSync } from 'node:fs';
  import { join } from 'node:path';
  
  const [root, defsPath] = process.argv.slice(2);
  const defs = JSON.parse(readFileSync(defsPath, 'utf8'));
  const tests = [
    'apps/server/test/merge-gate-digest.test.ts',
    'apps/server/test/merge-gate-paths.test.ts',
    'apps/server/test/merge-gate-evidence-link.test.ts',
  ];
  const diff = () => execFileSync('git', ['diff'], { cwd: root, encoding: 'utf8' });
  const before = diff();
  const rows = [];
  for (const [name, file, oldText, newText, all] of defs) {
    const path = join(root, file);
    const src = readFileSync(path, 'utf8');
    const count = src.split(oldText).length - 1;
    if (count === 0 || (!all && count !== 1)) {
      rows.push([name, `NOT APPLIED (${count} matches)`]);
      continue;
    }
    writeFileSync(path, all ? src.split(oldText).join(newText) : src.replace(oldText, () => newText));
    try {
      const r = spawnSync('pnpm', ['exec', 'vitest', 'run', ...tests], { cwd: root, encoding: 'utf8' });
      const out = `${r.stdout}\n${r.stderr}`.replace(/\x1b\[[0-9;]*m/g, '');
      const line = out.split('\n').find((l) => /^\s+Tests\s/.test(l)) ?? 'no summary';
      const failed = out
        .split('\n')
        .filter((l) => /^\s+× /.test(l))
        .map((l) => l.trim().replace(/ \d+ms$/, ''));
      rows.push([name, line.replace(/^\s+Tests\s+/, '').trim(), failed]);
    } finally {
      writeFileSync(path, src);
    }
  }
  const after = diff();
  for (const [name, summary] of rows) console.log(`${name.padEnd(48)} ${summary}`);
  console.log('');
  console.log(`git diff identical before and after the run: ${before === after}`);
  console.log('');
  console.log('failing tests per mutation:');
  for (const [name, , failed] of rows) {
    console.log(`- ${name}`);
    for (const f of failed ?? []) console.log(`    ${f}`);
  }
