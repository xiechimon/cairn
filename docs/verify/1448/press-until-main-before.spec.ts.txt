import { expect, type Locator, type Page, test } from '@playwright/test';

// pressUntil as it stands on origin/main 38dacbf (apps/web/e2e/helpers.ts).
async function pressUntil(page: Page, key: string, visible: Locator): Promise<void> {
  for (let attempt = 0; attempt < 6; attempt += 1) {
    await page.keyboard.press(key);
    const opened = await visible
      .waitFor({ state: 'visible', timeout: 1000 })
      .then(() => true)
      .catch(() => false);
    if (opened) return;
  }
  throw new Error(`${key} never brought up ${visible}`);
}

test('main: a strict-mode wait error is retried six times into a misleading message', async ({ page }) => {
  await page.goto('/app?scenario=01');
  await expect(page.getByRole('complementary')).toBeVisible();
  let presses = 0;
  await page.exposeFunction('countKey', () => {
    presses += 1;
  });
  await page.evaluate(() =>
    window.addEventListener('keydown', (e) => {
      if (e.key === 'F9') (window as unknown as { countKey: () => void }).countKey();
    }),
  );
  const err = await pressUntil(page, 'F9', page.locator('[data-testid]')).catch((e) => e);
  console.log('main strict ->', String(err).split('\n')[0], '| presses:', presses);
});
