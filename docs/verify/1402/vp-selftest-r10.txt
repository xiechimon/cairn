import { copyFileSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
import { spawnSync, execFileSync } from 'node:child_process';
import { join } from 'node:path';

const WT = '/home/user/pacman/.claude/worktrees/p1';
const SCRIPTS = join(WT, '.claude/skills/verify-pacman/scripts');
const VP = join(SCRIPTS, 'vp.mjs');
const BACKUP = process.argv[2];
const NODE = process.argv[3] ?? process.execPath;
const HEAD_TEST = join(SCRIPTS, 'vp-selftest-r9.mjs');
copyFileSync(VP, BACKUP);
writeFileSync(HEAD_TEST, execFileSync('git', ['-C', WT, 'show', '8568a96:.claude/skills/verify-pacman/scripts/vp-selftest.mjs']));

const original = readFileSync(VP, 'utf8');
function replaceOnce(src, from, to) {
  const n = src.split(from).length - 1;
  if (n !== 1) throw new Error(`expected one occurrence of ${from}, found ${n}`);
  return src.replace(from, to);
}

  ['M1', 'vp.mjs replaced by `process.exit(2);` (every command exits 2, prints nothing)', () => 'process.exit(2);\n'],

  ['M3', "--latest + --from/VERIFY_EVIDENCE_DIR guard message: 'cannot be combined with' -> 'excludes' (exit 2 kept)",

  ['M5', "EVIDENCE_ROOT = join(mainCheckout(SCRIPT_ROOT), ...) -> join(SCRIPT_ROOT, '.claude', 'verify-evidence')",
    () => replaceOnce(original, "const EVIDENCE_ROOT = join(mainCheckout(SCRIPT_ROOT), '.claude', 'verify-evidence');", "const EVIDENCE_ROOT = join(SCRIPT_ROOT, '.claude', 'verify-evidence');")],
];

  const r = spawnSync(NODE, [test], { cwd: WT, encoding: 'utf8' });
  const out = `${r.stdout}${r.stderr}`;
  return { code: r.status, fails: out.split('\n').filter((l) => l.startsWith('FAIL')), summary: out.trim().split('\n').at(-1) };
