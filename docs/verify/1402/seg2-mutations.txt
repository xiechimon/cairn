def build():
    root = os.path.join(SP, 'repo')
    shutil.rmtree(root, ignore_errors=True)
    os.makedirs(os.path.join(root, '.claude', 'skills'))
    os.makedirs(os.path.join(root, '.cursor', 'skills'))
    shutil.copytree(os.path.join(WT, SK), os.path.join(root, SK), symlinks=True)
    shutil.copytree(os.path.join(WT, '.cursor/skills/verify-pacman'), os.path.join(root, '.cursor/skills/verify-pacman'), symlinks=True)
    os.symlink(os.path.join(WT, 'node_modules'), os.path.join(root, 'node_modules'))
    st = os.path.join(root, S, 'vp-selftest.mjs')
    src = open(st).read()
    old = 'for (const [name, fn] of cases) {'
    assert src.count(old) == 1
    keep = json.dumps(NEW_CASES)
    src = src.replace(old, f'for (const [name, fn] of cases.filter(([n]) => {keep}.some((k) => n.includes(k)))) {{')
    src = src.replace('console.log(`${cases.length - failed}/${cases.length} passed`);', 'console.log(`ran ${failed} failed`);')
    open(st, 'w').write(src)
    return root

def mutate_registry(root, feature, case, need):
    p = os.path.join(root, SK, 'vp-registry.json')
    r = json.load(open(p))
    r['features'][feature]['cases'][case]['needs'].remove(need)
    open(p, 'w').write(json.dumps(r, indent=2) + '\n')

def run(root):
    t = time.time()
    out = subprocess.run([NODE, os.path.join(root, S, 'vp-selftest.mjs')], capture_output=True, text=True, cwd=root)
    fails = [l[6:] for l in out.stdout.splitlines() if l.startswith('FAIL  ')]
    passes = sum(1 for l in out.stdout.splitlines() if l.startswith('PASS  '))
    return fails, passes, round(time.time() - t)

only = sys.argv[1:] 
root = build()
fails, passes, secs = run(root)
print(f'baseline (no mutation): {passes} PASS, {len(fails)} FAIL {fails} [{secs}s]', flush=True)

for name, path, old, new in M:

    root = build()
    if name.startswith('G1b'):

    elif name.startswith('F-reg2'):
        mutate_registry(root, 'glass-journeys', 'create-task', 'own-stack')
    elif name.startswith('F-reg'):
        mutate_registry(root, 'api-keys', 'create', 'fresh-stack')
    else:
        p = os.path.join(root, path)
        s = open(p).read()

        open(p, 'w').write(s.replace(old, new))
    fails, passes, secs = run(root)
    caught += 1 if fails else 0
    print(f"{'RED  ' if fails else 'GREEN'} {name}: {len(fails)} FAIL [{secs}s]", flush=True)
    for f in fails:
        print(f'        FAIL  {f}', flush=True)
print(f'caught {caught} of {len([m for m in M if not only or any(m[0].startswith(o) for o in only)])}')
