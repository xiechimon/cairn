== the leg script (<scratchpad>/p53fix2-rerun.sh), also used for the re-runs appended to p5-3-fix-red-green.txt ==
  #!/usr/bin/env bash
  # Re-run the head step-attachments tests against the sources before a fix.
  # usage: p53fix2-rerun.sh <worktree> <scratch> <commit> <file>...
  set -u
  WT=$1; SC=$2; REV=$3; shift 3
  cd "$WT" || exit 2
  : > "$SC/sums-$REV.txt"
  for f in "$@"; do
    cp "$f" "$SC/$REV-$(basename "$f")"
    sha256sum "$f" >> "$SC/sums-$REV.txt"
  done
  restore() {
    for f in "$@"; do cp "$SC/$REV-$(basename "$f")" "$f"; done
  }
  trap 'restore "$@"' EXIT
  for f in "$@"; do git show "$REV:$f" > "$f"; done
  echo "sources replaced with $REV: $*"
  NO_COLOR=1 FORCE_COLOR=0 pnpm exec vitest run --reporter=verbose apps/daemon/test/step-attachments.test.ts > "$SC/log-$REV.txt" 2>&1
  echo "vitest exit: $?"
  restore "$@"
  trap - EXIT
  sha256sum -c "$SC/sums-$REV.txt"
