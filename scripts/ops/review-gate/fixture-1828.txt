function workCall(
  call: ToolCallRecord,
  fallbackSeconds: number | null,
  stepRunning: boolean,
): WorkStep {
  const args = (call.arguments ?? {}) as Record<string, unknown>;
  const name = call.name.toLowerCase();
  const path = String(args.file_path ?? args.path ?? '').replace(
    /^\/(?:home|Users)\/[^/]+\/\.(?:pacman-chief|cairn)\/(?:workspaces|plans)\/[^/]+\//,
    '',
  );
  const output = resultToText(call.result);
  const seconds =
    call.startedAt !== undefined && call.endedAt !== undefined
      ? Math.max(0, (call.endedAt - call.startedAt) / 1000)
      : fallbackSeconds;
  const base = {
    kind: 'call' as const,
    id: call.id,
    label: call.name,
    output: output !== null && isJsonPlumbing(output) ? null : output,
    error: call.isError === true,
    running: call.result === undefined && stepRunning,
    seconds,
  };
  if (name === 'read') {
    const offset = typeof args.offset === 'number' ? args.offset : null;
    const limit = typeof args.limit === 'number' && args.limit > 0 ? args.limit : null;
    const range =
      offset === null && limit === null
        ? ''
        : limit === null
          ? ` L${offset}+`
          : ` L${offset ?? 1}–${(offset ?? 1) + limit - 1}`;
    return { ...base, action: 'read', target: path + range };
  }
  if (name === 'edit' || name === 'write' || name === 'multiedit') {
    const edits = name === 'multiedit' && Array.isArray(args.edits) ? args.edits : [args];
    const removed: string[] = [];
    const added: string[] = [];
    for (const edit of edits) {
      if (edit === null || typeof edit !== 'object') continue;
      const values = edit as Record<string, unknown>;
      const oldText = values.old_string ?? values.oldText;
      const newText = values.new_string ?? values.newText ?? values.content;
      if (typeof oldText === 'string' && oldText !== '') removed.push(...oldText.split('\n'));
      if (typeof newText === 'string' && newText !== '') added.push(...newText.split('\n'));
    }
    return {
      ...base,
      action: 'edit',
      target: path,
      diff: { added: added.length, removed: removed.length },
      changes: { removed, added },
    };
  }
  if (name === 'grep' || name === 'find' || name === 'ls' || name === 'glob') {
    const pattern = typeof args.pattern === 'string' ? args.pattern : '';
    return {
      ...base,
      action: 'search',
      description: [call.name, pattern, path].filter((part) => part !== '').join(' '),
    };
  }
  if (name === 'bash') {
    const command = typeof args.command === 'string' ? args.command : '';
    const description =
      typeof args.description === 'string' ? args.description : (command.split('\n')[0] ?? '');
    return {
      ...base,
      action: /^(grep|rg|find|git grep)\b/.test(command.trim()) ? 'search' : 'run',
      description,
      command,
    };
  }
  return { ...base, action: 'tool', target: call.name.replace(/^mcp__/, '').replace(/__/g, ' · ') };
}

function resultToText(result: unknown): string | null {
  if (result == null) return null;
  if (typeof result === 'string') return result === '' ? null : result;
  if (Array.isArray(result)) {
    const joined = result
      .map(resultToText)
      .filter((s): s is string => s != null && s !== '')
      .join('\n');
    return joined === '' ? null : joined;
  }
  if (typeof result === 'object') {
    const o = result as Record<string, unknown>;
    if (typeof o.text === 'string') return o.text === '' ? null : o.text;
    for (const key of ['content', 'output', 'stdout', 'result'] as const) {
      if (o[key] !== undefined) {
        const nested = resultToText(o[key]);
        if (nested != null && nested !== '') return nested;
      }
    }
  }
  return null;
}

function isJsonPlumbing(text: string): boolean {
  const trimmed = text.trim();
  if (!trimmed.startsWith('{') && !trimmed.startsWith('[')) return false;
  try {
    JSON.parse(trimmed);
    return true;
  } catch {
    return false;
  }
}

export function acpErrorMessage(error: unknown): string {
  const message = error instanceof Error ? error.message : String(error);
  if (/spawn gemini .*ENOENT|gemini.*not found/i.test(message)) {
    return 'Gemini executable not found: install Gemini CLI with npm install -g @google/gemini-cli';
  }
  if (
    /authenticate|required|not logged in|login|api key is missing|auth method|GEMINI_API_KEY/i.test(
      message,
    )
  ) {
    return `Gemini login failed: run gemini and complete its login flow. ${message}`;
  }
  return message;
}

export function mapAcpUpdate(update: SessionUpdate | { kind: 'cancelled' }): StepEvent | null {
  if ('kind' in update)
    return { type: 'message_stop', message: { role: 'assistant', content: '' } };
  if (update.sessionUpdate === 'agent_message_chunk' && update.content.type === 'text') {
    return { type: 'text_delta', text: update.content.text };
  }
  if (update.sessionUpdate === 'tool_call' || update.sessionUpdate === 'tool_call_update') {
    return {
      type: 'toolcall_end',
      call: {
        id: update.toolCallId,
        name: update.name ?? update.title ?? 'unknown',
        arguments: 'rawInput' in update ? (update.rawInput ?? {}) : {},
        ...(update.status === 'failed' ? { isError: true } : {}),
      },
    };
  }
  return null;
}
